# JAN AI — Full Backend Implementation (FastAPI + SQLAlchemy + APScheduler) This document provides the complete, modular backend code for the JAN AI - Automation Engine. ## 1. Directory Structure ```text backend/ ├── app/ │ ├── main.py # Application entry point │ ├── database.py # SQLAlchemy engine & session │ ├── models.py # DB Schema (Accounts, Tasks, Logs, Content) │ ├── schemas.py # Pydantic models for API validation │ ├── routes/ │ │ ├── accounts.py # Multi-brand management │ │ ├── pipeline.py # Content transformation engine │ │ ├── tasks.py # Scheduler & task CRUD │ │ └── suggestions.py # Rule-based insights │ ├── services/ │ │ ├── content_engine.py # Extraction & Formatting logic │ │ ├── strategy_engine.py # Weekly plan generation │ │ └── scheduler_service.py # APScheduler integration │ └── config.py # Environment variables ├── requirements.txt └── run.py ``` ## 2. Core Models (app/models.py) ```python from sqlalchemy import Column, Integer, String, DateTime, ForeignKey, JSON, Text, Boolean from .database import Base import datetime class Account(Base): __tablename__ = "accounts" id = Column(Integer, primary_key=True, index=True) name = Column(String) niche = Column(String) audience = Column(String) goal = Column(String) tone = Column(String, default="Professional") class ContentEntry(Base): __tablename__ = "content_entries" id = Column(Integer, primary_key=True, index=True) account_id = Column(Integer, ForeignKey("accounts.id")) raw_input = Column(Text) topics = Column(JSON) intent = Column(String) generated_posts = Column(JSON) # Multi-format data scores = Column(JSON) created_at = Column(DateTime, default=datetime.datetime.utcnow) class WeeklyPlan(Base): __tablename__ = "weekly_plans" id = Column(Integer, primary_key=True, index=True) account_id = Column(Integer, ForeignKey("accounts.id")) plan_data = Column(JSON) # 7-day breakdown is_active = Column(Boolean, default=True) created_at = Column(DateTime, default=datetime.datetime.utcnow) class ExecutionLog(Base): __tablename__ = "execution_logs" id = Column(Integer, primary_key=True, index=True) task_id = Column(String) event_type = Column(String) status = Column(String) # Success, Error, Pending message = Column(Text) timestamp = Column(DateTime, default=datetime.datetime.utcnow) ``` ## 3. Main Application Entry (app/main.py) ```python from fastapi import FastAPI from app.routes import pipeline, accounts, tasks, suggestions from app.services.scheduler_service import start_scheduler app = FastAPI(title="JAN AI - Automation Engine") # Include Routers app.include_router(accounts.router, prefix="/accounts", tags=["Accounts"]) app.include_router(pipeline.router, prefix="/pipeline", tags=["Pipeline"]) app.include_router(tasks.router, prefix="/tasks", tags=["Tasks"]) app.include_router(suggestions.router, prefix="/suggestions", tags=["Suggestions"]) @app.on_event("startup") async def startup_event(): start_scheduler() @app.get("/health") def health_check(): return {"status": "active", "engine": "v2.4.0", "db": "connected"} ```
